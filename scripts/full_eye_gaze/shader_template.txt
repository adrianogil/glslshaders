// FULL-IMAGE neural eye with learned TWO-AXIS gaze.
// Every RGB pixel is predicted by the 28,995-parameter network below.
// No image rotation/warp, analytic eye rendering, textures, or cached animation.
// One Shadertoy Image pass; leave all channels empty.
// Drag X/Y to look horizontally/vertically. Release to resume wide saccades.
// Hue and pupil are also neural inputs. Change the constants below to tune them.
// Trained yaw 32–76 degrees; pitch -28–22 degrees. Not an arbitrary-camera field.
precision highp float;

const float IRIS_HUE = .56;
const float PUPIL_SIZE = .40;

__NETWORK__

// Each fixation is genuinely rendered at a learned (yaw,pitch), not screen-warped.
vec2 saccadeGaze(float time) {
  float t=mod(time,10.0);
  // Wide opposing corners mixed with horizontal, vertical, and corrective jumps.
  vec4 endpoints=vec4(.48,.46,.08,.16);
  vec2 window=vec2(0.,1.25);
  if(t>=1.25){endpoints=vec4(.08,.16,.92,.84);window=vec2(1.25,2.25);}
  if(t>=2.25){endpoints=vec4(.92,.84,.12,.84);window=vec2(2.25,3.50);}
  if(t>=3.50){endpoints=vec4(.12,.84,.12,.12);window=vec2(3.50,4.40);}
  if(t>=4.40){endpoints=vec4(.12,.12,.88,.72);window=vec2(4.40,5.90);}
  if(t>=5.90){endpoints=vec4(.88,.72,.82,.20);window=vec2(5.90,6.85);}
  if(t>=6.85){endpoints=vec4(.82,.20,.18,.76);window=vec2(6.85,8.20);}
  if(t>=8.20){endpoints=vec4(.18,.76,.48,.46);window=vec2(8.20,10.0);}
  float age=t-window.x;
  vec2 delta=endpoints.zw-endpoints.xy;
  float flight=.065+.055*length(delta);
  float u=clamp(age/flight,0.,1.);
  float ease=u*u*u*(u*(u*6.-15.)+10.);
  float tail=clamp((age-flight)/.16,0.,1.);
  vec2 settle=.025*delta*sin(6.28318530718*tail)*(1.-tail)*(1.-tail);
  float hold=smoothstep(.26,.42,age)*(1.-smoothstep(window.y-window.x-.16,window.y-window.x,age));
  vec2 drift=.0007*hold*sin(vec2(11.3,14.9)*age);
  return clamp(mix(endpoints.xy,endpoints.zw,ease)+settle+drift,0.,1.);
}

vec4 eyeConditions() {
  vec2 gaze=saccadeGaze(iTime);
  float phase=6.28318530718*mod(iTime,10.0)/10.0;
  vec4 controls=vec4(gaze.x,clamp(PUPIL_SIZE+.12*sin(phase+.8),0.,1.),IRIS_HUE,gaze.y);
  // Positive geometric pitch looks down, so upper mouse positions use low pitch.
  if(iMouse.z>0.){vec2 mouse=clamp(iMouse.xy/iResolution.xy,0.,1.);controls.x=mouse.x;controls.w=1.-mouse.y;}
#ifdef FULL_GAZE_LOCAL
  controls.y=uControls.y;controls.z=uControls.z;
  if(uManual==1)controls=uControls;
  if(uProbeMode==1)controls=uProbeControls;
#endif
  return controls;
}

void mainImage(out vec4 fragColor,in vec2 fragCoord) {
  vec2 p=(2.*fragCoord-iResolution.xy)/min(iResolution.x,iResolution.y);
  p=clamp(p,vec2(-1.4),vec2(1.4));
#ifdef FULL_GAZE_LOCAL
  if(uProbeMode==1)p=uProbePosition;
#endif
  fragColor=vec4(neuralFullEye(p,eyeConditions()),1.);
}
